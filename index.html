-- 1. Payment / Capital Accounts Table (Cash, Bank, Capital) CREATE TABLE payment_accounts ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, name TEXT NOT NULL, -- e.g., 'Cash in Hand', 'Meezan Bank', 'Capital' account_type TEXT CHECK (account_type IN ('Cash', 'Bank', 'Capital')), balance NUMERIC DEFAULT 0, created_at TIMESTAMPTZ DEFAULT NOW() ); -- Default Cash and Bank Accounts INSERT INTO payment_accounts (name, account_type, balance) VALUES ('Cash Wallet', 'Cash', 0), ('Main Bank Account', 'Bank', 0); -- 2. Products / Inventory Table CREATE TABLE products ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, name TEXT NOT NULL, purchase_price NUMERIC DEFAULT 0, sale_price NUMERIC DEFAULT 0, stock_quantity NUMERIC DEFAULT 0, created_at TIMESTAMPTZ DEFAULT NOW() ); -- 3. Existing Accounts Table CREATE TABLE IF NOT EXISTS accounts ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, name TEXT NOT NULL, category TEXT CHECK (category IN ('Employee', 'Customer', 'Supplier', 'Office')), base_salary NUMERIC DEFAULT 0, created_at TIMESTAMPTZ DEFAULT NOW() ); -- 4. Expanded Transactions Ledger CREATE TABLE IF NOT EXISTS transactions ( id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY, account_id BIGINT REFERENCES accounts(id) ON DELETE SET NULL, payment_account_id BIGINT REFERENCES payment_accounts(id) ON DELETE SET NULL, product_id BIGINT REFERENCES products(id) ON DELETE SET NULL, type TEXT CHECK (type IN ('Payment_In', 'Payment_Out', 'Salary', 'Bonus', 'Commission', 'Credit', 'Product_Purchase', 'Product_Sale')), amount NUMERIC DEFAULT 0, quantity NUMERIC DEFAULT 0, notes TEXT, created_at TIMESTAMPTZ DEFAULT NOW() );
