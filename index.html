Production MySQL 8.0 Schema Architecture
Complete relational database schema designed specifically for Ufone Telecom Franchise double-entry accounting. 
-- =========================================================================
-- HASNAIN COMMUNICATION - TELECOM & FRANCHISE ACCOUNTING SYSTEM SCHEMA
-- TARGET DATABASE: MySQL 8.0+ / MariaDB 10.5+
-- ENGINE: InnoDB | CHARSET: utf8mb4_unicode_ci
-- =========================================================================

CREATE DATABASE IF NOT EXISTS `hasnain_telecom_db` 
DEFAULT CHARACTER SET utf8mb4 
COLLATE utf8mb4_unicode_ci;

USE `hasnain_telecom_db`;

-- 1. PAYMENT ACCOUNTS (CASH & BANK BALANCES)
CREATE TABLE `payment_accounts` (
  `account_id` INT AUTO_INCREMENT PRIMARY KEY,
  `account_name` VARCHAR(100) NOT NULL,
  `account_type` ENUM('Cash', 'Bank', 'Mobile Wallet') NOT NULL,
  `account_number` VARCHAR(50) DEFAULT NULL,
  `bank_name` VARCHAR(100) DEFAULT NULL,
  `current_balance` DECIMAL(15,2) NOT NULL DEFAULT 0.00,
  `status` ENUM('Active', 'Inactive') DEFAULT 'Active',
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- 2. CHART OF ACCOUNTS (DOUBLE-ENTRY LEDGER)
CREATE TABLE `chart_of_accounts` (
  `account_code` VARCHAR(20) PRIMARY KEY,
  `account_name` VARCHAR(100) NOT NULL,
  `account_group` ENUM('Asset', 'Liability', 'Equity', 'Revenue', 'Expense') NOT NULL,
  `opening_balance` DECIMAL(15,2) DEFAULT 0.00,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- 3. RETAILERS & CUSTOMERS (UDHAR MANAGEMENT)
CREATE TABLE `retailers_customers` (
  `retailer_id` INT AUTO_INCREMENT PRIMARY KEY,
  `shop_name` VARCHAR(150) NOT NULL,
  `owner_name` VARCHAR(100) NOT NULL,
  `phone_number` VARCHAR(20) NOT NULL UNIQUE,
  `address` VARCHAR(255) DEFAULT NULL,
  `credit_limit` DECIMAL(12,2) DEFAULT 50000.00,
  `current_udhar_balance` DECIMAL(12,2) DEFAULT 0.00,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- 4. VENDORS & SUPPLIERS (MAIN TELECOM COMPANIES)
CREATE TABLE `vendors_suppliers` (
  `vendor_id` INT AUTO_INCREMENT PRIMARY KEY,
  `company_name` VARCHAR(150) NOT NULL,
  `contact_person` VARCHAR(100) DEFAULT NULL,
  `phone_number` VARCHAR(20) DEFAULT NULL,
  `current_payable_balance` DECIMAL(15,2) DEFAULT 0.00,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- 5. ACCOUNTING VOUCHERS (HEAD TABLE)
CREATE TABLE `vouchers` (
  `voucher_id` BIGINT AUTO_INCREMENT PRIMARY KEY,
  `voucher_no` VARCHAR(30) NOT NULL UNIQUE,
  `voucher_type` ENUM('Receipt', 'Payment', 'Journal', 'Contra') NOT NULL,
  `voucher_date` DATE NOT NULL,
  `total_amount` DECIMAL(15,2) NOT NULL,
  `narration` TEXT DEFAULT NULL,
  `posted_by` INT DEFAULT 1,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;

-- 6. VOUCHER ITEMS (DOUBLE-ENTRY LINE ITEMS)
CREATE TABLE `voucher_items` (
  `item_id` BIGINT AUTO_INCREMENT PRIMARY KEY,
  `voucher_id` BIGINT NOT NULL,
  `account_code` VARCHAR(20) NOT NULL,
  `debit_amount` DECIMAL(15,2) DEFAULT 0.00,
  `credit_amount` DECIMAL(15,2) DEFAULT 0.00,
  FOREIGN KEY (`voucher_id`) REFERENCES `vouchers`(`voucher_id`) ON DELETE CASCADE,
  FOREIGN KEY (`account_code`) REFERENCES `chart_of_accounts`(`account_code`)
) ENGINE=InnoDB;

-- 7. TELECOM SALES TRANSACTIONS
CREATE TABLE `telecom_sales` (
  `sale_id` BIGINT AUTO_INCREMENT PRIMARY KEY,
  `sale_type` ENUM('Easyload', 'SIM_New', 'SIM_Swap', 'Bundle', 'Utility_Bill') NOT NULL,
  `retailer_id` INT DEFAULT NULL,
  `gross_amount` DECIMAL(12,2) NOT NULL,
  `commission_earned` DECIMAL(10,2) NOT NULL DEFAULT 0.00,
  `payment_account_id` INT NOT NULL,
  `remarks` VARCHAR(255) DEFAULT NULL,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (`retailer_id`) REFERENCES `retailers_customers`(`retailer_id`),
  FOREIGN KEY (`payment_account_id`) REFERENCES `payment_accounts`(`account_id`)
) ENGINE=InnoDB;

-- 8. DAILY EXPENSES
CREATE TABLE `daily_expenses` (
  `expense_id` INT AUTO_INCREMENT PRIMARY KEY,
  `category` VARCHAR(50) NOT NULL,
  `amount` DECIMAL(10,2) NOT NULL,
  `description` VARCHAR(255) NOT NULL,
  `paid_from_account_id` INT NOT NULL,
  `expense_date` DATE NOT NULL,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
) ENGINE=InnoDB;
