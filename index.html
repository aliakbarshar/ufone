/* =================================================================
   1. MYSQL DATABASE SCHEMA (schema.sql)
   ================================================================= */

CREATE DATABASE IF NOT EXISTS apexbank_accounting CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE apexbank_accounting;

-- Users & Roles Table
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    role ENUM('Super Admin', 'Accountant', 'Cashier', 'Manager') DEFAULT 'Accountant',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Chart of Accounts
CREATE TABLE chart_of_accounts (
    id INT AUTO_INCREMENT PRIMARY KEY,
    code VARCHAR(20) UNIQUE NOT NULL,
    title VARCHAR(150) NOT NULL,
    type ENUM('Asset', 'Liability', 'Equity', 'Revenue', 'Expense') NOT NULL
);

-- Customers Table
CREATE TABLE customers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    customer_code VARCHAR(20) UNIQUE NOT NULL,
    name VARCHAR(150) NOT NULL,
    phone VARCHAR(50),
    city VARCHAR(50),
    balance DECIMAL(15, 2) DEFAULT 0.00,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Suppliers Table
CREATE TABLE suppliers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    vendor_code VARCHAR(20) UNIQUE NOT NULL,
    name VARCHAR(150) NOT NULL,
    company_name VARCHAR(150),
    phone VARCHAR(50),
    payable_balance DECIMAL(15, 2) DEFAULT 0.00,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Vouchers & General Ledger Transactions
CREATE TABLE vouchers (
    id INT AUTO_INCREMENT PRIMARY KEY,
    voucher_num VARCHAR(30) UNIQUE NOT NULL,
    voucher_date DATE NOT NULL,
    voucher_type ENUM('Payment', 'Receipt', 'Journal', 'Contra') NOT NULL,
    debit_account_id INT,
    credit_account_id INT,
    amount DECIMAL(15, 2) NOT NULL,
    description TEXT,
    created_by INT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (created_by) REFERENCES users(id)
);


/* =================================================================
   2. PHP MVC CONTROLLER EXAMPLE (VoucherController.php)
   ================================================================= */

<?php
namespace Controllers;

use Models\VoucherModel;
use Core\Controller;

class VoucherController extends Controller {

    private $voucherModel;

    public function __construct() {
        $this->voucherModel = new VoucherModel();
    }

    public function createVoucher() {
        // CSRF Verification
        if ($_SERVER['REQUEST_METHOD'] === 'POST') {
            if (!verify_csrf_token($_POST['csrf_token'])) {
                die(json_encode(['status' => 'error', 'message' => 'CSRF Token validation failed']));
            }

            $data = [
                'voucher_num' => 'VOUCH-' . time(),
                'voucher_date' => $_POST['voucher_date'],
                'voucher_type' => $_POST['voucher_type'],
                'amount' => floatval($_POST['amount']),
                'description' => htmlspecialchars($_POST['description'])
            ];

            $result = $this->voucherModel->saveVoucher($data);
            echo json_encode(['status' => 'success', 'data' => $result]);
        }
    }
}
